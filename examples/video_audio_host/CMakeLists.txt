cmake_minimum_required(VERSION 3.20)
project(cui_video_audio_foundation LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIDEO_AUDIO_SANITIZE "Build the foundation with ASan/UBSan" OFF)
find_package(Threads REQUIRED)
add_library(video_audio_host STATIC host.c)
target_include_directories(video_audio_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(video_audio_host PUBLIC Threads::Threads)
add_executable(test_video_audio_host test_host.c)
target_link_libraries(test_video_audio_host PRIVATE video_audio_host)
if(NOT WIN32)
 target_link_libraries(video_audio_host PUBLIC m)
endif()
foreach(target video_audio_host test_video_audio_host)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /UNDEBUG)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  if(VIDEO_AUDIO_SANITIZE)
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
 endif()
endforeach()
enable_testing()
foreach(case accounting backpressure generations failures cancellation concurrent)
 add_test(NAME video_audio_${case} COMMAND test_video_audio_host ${case})
 set_tests_properties(video_audio_${case} PROPERTIES TIMEOUT 20)
endforeach()

# Measurement only: never a hardware-latency or hard realtime CTest assertion.
if(UNIX)
 add_executable(bench_video_audio_mutex bench_mutex.c)
 add_test(NAME video_audio_mutex_observation COMMAND bench_video_audio_mutex)
 set_tests_properties(video_audio_mutex_observation PROPERTIES TIMEOUT 20 LABELS benchmark)
 target_link_libraries(bench_video_audio_mutex PRIVATE video_audio_host)
 target_compile_options(bench_video_audio_mutex PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 if(VIDEO_AUDIO_SANITIZE)
  target_compile_options(bench_video_audio_mutex PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 endif()
endif()
