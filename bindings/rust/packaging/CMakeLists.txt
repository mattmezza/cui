cmake_minimum_required(VERSION 3.20)
project(cui_rust_package LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CUI_BUILD_BIDI ${CUI_BUILD_NATIVE} CACHE BOOL "" FORCE)
set(CUI_BUILD_CPP_EXAMPLE OFF CACHE BOOL "" FORCE)
set(CUI_BUILD_BRIDGE OFF CACHE BOOL "" FORCE)
if(NOT DEFINED CUI_PACKAGE_ROOT)
 get_filename_component(CUI_PACKAGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
endif()
add_subdirectory("${CUI_PACKAGE_ROOT}" cui_source EXCLUDE_FROM_ALL)
add_library(cui_rust_bridge STATIC "${CUI_PACKAGE_ROOT}/src/bridge/cui_bridge.c")
target_include_directories(cui_rust_bridge PRIVATE "${CUI_PACKAGE_ROOT}/include" "${CUI_PACKAGE_ROOT}/src/bridge")
target_link_libraries(cui_rust_bridge PRIVATE cui)
if(CUI_BUILD_NATIVE)
 if(NOT TARGET cui_sdl3)
  message(FATAL_ERROR "sdl feature requires SDL3 >=3.2, FreeType, HarfBuzz development packages")
 endif()
 target_sources(cui_rust_bridge PRIVATE "${CUI_PACKAGE_ROOT}/src/bridge/cui_bridge_sdl.c")
 target_compile_definitions(cui_rust_bridge PRIVATE CUI_BRIDGE_WITH_SDL=1)
 target_link_libraries(cui_rust_bridge PRIVATE cui_sdl3)
endif()
target_compile_options(cui_rust_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
