cmake_minimum_required(VERSION 3.20)
project(cui_native_starter LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CUI REQUIRED cui-sdl3)
# CUI installs static archives; retain Requires.private and Libs.private.
add_library(starter_app STATIC app.c)
target_include_directories(starter_app PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CUI_STATIC_INCLUDE_DIRS})
target_compile_options(starter_app PUBLIC ${CUI_STATIC_CFLAGS_OTHER})
target_link_directories(starter_app PUBLIC ${CUI_STATIC_LIBRARY_DIRS})
target_link_libraries(starter_app PUBLIC ${CUI_STATIC_LIBRARIES})
target_link_options(starter_app PUBLIC ${CUI_STATIC_LDFLAGS_OTHER})
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(starter_app PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(native_starter main.c)
target_link_libraries(native_starter PRIVATE starter_app)
add_executable(test_starter_app test_app.c)
target_link_libraries(test_starter_app PRIVATE starter_app)
enable_testing()
add_test(NAME starter_interactions COMMAND test_starter_app)
