cmake_minimum_required(VERSION 3.20)
project(video_audio_ffmpeg LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIDEO_AUDIO_SANITIZE "ASan and UBSan" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FF REQUIRED IMPORTED_TARGET libavformat=63.1.102 libavcodec=63.1.102 libavutil=61.1.102 libswresample=7.1.102)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(decoder STATIC decoder.c ../video_audio_host/host.c)
target_link_libraries(decoder PUBLIC PkgConfig::FF Threads::Threads m)
add_executable(test_decoder test_decoder.c)
target_link_libraries(test_decoder PRIVATE decoder)
foreach(t decoder test_decoder)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 if(VIDEO_AUDIO_SANITIZE)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
enable_testing()
add_test(NAME fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/fixtures.py ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(fixtures PROPERTIES FIXTURES_SETUP media TIMEOUT 30)
foreach(case pcm resample offset limits cancel controller stale_shutdown unsupported)
 add_test(NAME ffmpeg_${case} COMMAND test_decoder ${case} ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
 set_tests_properties(ffmpeg_${case} PROPERTIES FIXTURES_REQUIRED media TIMEOUT 30)
endforeach()
