cmake_minimum_required(VERSION 3.20)
project(cui_video_sound LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIDEO_SOUND_SANITIZE "ASan/UBSan" OFF)
option(VIDEO_SOUND_UI "Build the actual CUI native sound player and private smoke" OFF)
get_filename_component(ROOT ../.. ABSOLUTE BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
pkg_check_modules(SOUND REQUIRED IMPORTED_TARGET sdl3=3.4.18 libavformat=63.1.102 libavcodec=63.1.102 libavutil=61.1.102 libswscale=10.1.102 libswresample=7.1.102)
add_library(video_sound STATIC session.c ../video_player/decoder.c ../video_audio_host/host.c ../video_audio_ffmpeg/decoder.c ../video_audio_sdl/output.c ../video_audio_sdl/sdl.c)
target_include_directories(video_sound PUBLIC ${ROOT}/include)
target_link_libraries(video_sound PUBLIC PkgConfig::SOUND Threads::Threads m)
add_executable(test_sound test_session.c)
target_link_libraries(test_sound PRIVATE video_sound)
add_executable(test_sound_watch test_watch.c)
target_link_libraries(test_sound_watch PRIVATE video_sound)
add_executable(test_sound_output ../video_audio_sdl/test_output.c)
target_link_libraries(test_sound_output PRIVATE video_sound)
set(OWN_TARGETS video_sound test_sound test_sound_watch test_sound_output)
enable_testing()
add_test(NAME sound_fixture COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/make_fixture.py ${CMAKE_CURRENT_BINARY_DIR}/media)
set_tests_properties(sound_fixture PROPERTIES FIXTURES_SETUP sound_media TIMEOUT 30)
foreach(case combined preview origin failures endpoints missing invalidation)
 add_test(NAME sound_${case} COMMAND test_sound ${case} ${CMAKE_CURRENT_BINARY_DIR}/media)
 set_tests_properties(sound_${case} PROPERTIES FIXTURES_REQUIRED sound_media TIMEOUT 20 ENVIRONMENT SDL_AUDIODRIVER=dummy)
endforeach()
add_test(NAME sound_watch COMMAND test_sound_watch)
set_tests_properties(sound_watch PROPERTIES TIMEOUT 30 ENVIRONMENT SDL_AUDIODRIVER=dummy)
foreach(case drain silence pause generations failures units teardown dummy unbound setup)
 add_test(NAME sound_output_${case} COMMAND test_sound_output ${case})
 set_tests_properties(sound_output_${case} PROPERTIES TIMEOUT 20 ENVIRONMENT SDL_AUDIODRIVER=dummy)
endforeach()
if(VIDEO_SOUND_UI)
 set(CUI_REQUIRE_NATIVE ON CACHE BOOL "" FORCE)
 set(CUI_FETCH_SDL OFF CACHE BOOL "" FORCE)
 set(CUI_SANITIZE ${VIDEO_SOUND_SANITIZE} CACHE BOOL "" FORCE)
 add_subdirectory(${ROOT} cui)
 add_executable(video_player_audio main.c)
 target_link_libraries(video_player_audio PRIVATE video_sound cui_sdl3)
 add_executable(test_sound_legacy ../video_player/test_decoder.c)
 target_link_libraries(test_sound_legacy PRIVATE video_sound cui_software)
 list(APPEND OWN_TARGETS video_player_audio test_sound_legacy)
 add_test(NAME sound_legacy_video COMMAND test_sound_legacy ${CMAKE_CURRENT_BINARY_DIR}/media/fixture.mkv)
 set_tests_properties(sound_legacy_video PROPERTIES FIXTURES_REQUIRED sound_media TIMEOUT 20)
 add_test(NAME sound_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_native.py $<TARGET_FILE:video_player_audio> ${ROOT}/third_party/inter/Inter-Regular.otf ${CMAKE_CURRENT_BINARY_DIR}/media/fixture.mkv)
 set_tests_properties(sound_native PROPERTIES FIXTURES_REQUIRED sound_media TIMEOUT 30 SKIP_RETURN_CODE 77)
 add_test(NAME sound_native_failures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_native_failures.py $<TARGET_FILE:video_player_audio> ${ROOT}/third_party/inter/Inter-Regular.otf ${CMAKE_CURRENT_BINARY_DIR}/media/fixture.mkv)
 set_tests_properties(sound_native_failures PROPERTIES FIXTURES_REQUIRED sound_media TIMEOUT 60 SKIP_RETURN_CODE 77)
endif()
foreach(t ${OWN_TARGETS})
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 if(VIDEO_SOUND_SANITIZE)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
