cmake_minimum_required(VERSION 3.20)
project(cui_speech_input_example LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(CUI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3>=3.2)
add_library(speech_input_host STATIC host.c host_sdl.c)
target_include_directories(speech_input_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CUI_ROOT}/include")
target_link_libraries(speech_input_host PRIVATE PkgConfig::SDL3)
add_executable(test_host test_host.c "${CUI_ROOT}/src/base/cui_speech.c")
target_link_libraries(test_host PRIVATE speech_input_host)
if(NOT WIN32)
  target_link_libraries(test_host PRIVATE m)
endif()
add_executable(test_conversion "${CUI_ROOT}/docs/design/speech_sdl_conversion_probe.c")
target_link_libraries(test_conversion PRIVATE PkgConfig::SDL3)
foreach(target speech_input_host test_host test_conversion)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME speech_input_host_offline COMMAND test_host)
add_test(NAME speech_input_conversion_offline COMMAND test_conversion)

# GNU/Clang ELF wrapping prevents every device operation in this fixture.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(test_sdl test_sdl.c)
  target_link_libraries(test_sdl PRIVATE speech_input_host PkgConfig::SDL3)
  target_compile_options(test_sdl PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(symbol SDL_WasInit SDL_OpenAudioDeviceStream SDL_ResumeAudioStreamDevice SDL_SetAudioStreamPutCallback SDL_DestroyAudioStream)
    target_link_options(test_sdl PRIVATE "LINKER:--wrap=${symbol}")
  endforeach()
  add_test(NAME speech_input_sdl_injected COMMAND test_sdl)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Threads REQUIRED)
  add_executable(test_thread test_thread.c "${CUI_ROOT}/src/base/cui_speech.c")
  target_link_libraries(test_thread PRIVATE speech_input_host Threads::Threads m)
  target_compile_options(test_thread PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME speech_input_threaded_offline COMMAND test_thread)
  set_tests_properties(speech_input_threaded_offline PROPERTIES TIMEOUT 10)
endif()
