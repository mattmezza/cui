cmake_minimum_required(VERSION 3.20)
project(cui_language_examples LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CUI_EXAMPLE_NATIVE "Enable SDL3 windows when dependencies are available" ON)
set(CUI_BUILD_NATIVE ${CUI_EXAMPLE_NATIVE} CACHE BOOL "" FORCE)
add_subdirectory(../.. cui-build)
add_library(cui_example_host SHARED host.c)
target_link_libraries(cui_example_host PRIVATE cui)
target_include_directories(cui_example_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
cui_warnings(cui_example_host)
if(TARGET cui_sdl3)
 target_link_libraries(cui_example_host PRIVATE cui_sdl3)
 target_compile_definitions(cui_example_host PRIVATE CUI_EXAMPLE_NATIVE=1)
 message(STATUS "Language examples: native SDL3 window enabled")
else()
 message(STATUS "Language examples: headless only; native run returns 4")
endif()
find_path(NODE_API_INCLUDE_DIR node_api.h PATHS /usr/include/node)
if(NODE_API_INCLUDE_DIR)
 add_library(cui_node MODULE node/addon.c)
 target_include_directories(cui_node PRIVATE ${NODE_API_INCLUDE_DIR})
 target_link_libraries(cui_node PRIVATE cui_example_host)
 set_target_properties(cui_node PROPERTIES PREFIX "" OUTPUT_NAME "cui" SUFFIX ".node" BUILD_RPATH "$ORIGIN")
 cui_warnings(cui_node)
endif()
add_executable(cui_example_host_test host_test.c)
target_link_libraries(cui_example_host_test PRIVATE cui_example_host)
cui_warnings(cui_example_host_test)
