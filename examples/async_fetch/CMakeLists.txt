cmake_minimum_required(VERSION 3.20)
project(cui_async_fetch_example LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This optional POSIX host example is currently supported/tested on Linux only")
endif()
option(AF_BUILD_UI "Build the actual SDL3 example and offline UI test" ON)
option(AF_WITH_CURL "Build the optional libcurl HTTPS transport" ON)
# No downloads. Explicit version overrides require local revalidation.
set(AF_SDL_VERSION "3.4.18" CACHE STRING "Exact SDL pkg-config version validated by this example")
set(AF_CURL_VERSION "8.22.0" CACHE STRING "Exact libcurl pkg-config version validated by this example")
find_package(Threads REQUIRED)
function(af_warnings target)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endfunction()
add_library(async_fetch_worker STATIC fetch.c)
target_link_libraries(async_fetch_worker PUBLIC Threads::Threads)
target_include_directories(async_fetch_worker PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
af_warnings(async_fetch_worker)
enable_testing()
add_executable(test_async_fetch test_fetch.c)
target_link_libraries(test_async_fetch PRIVATE async_fetch_worker)
af_warnings(test_async_fetch)
add_test(NAME async_fetch_worker COMMAND test_async_fetch)
set_tests_properties(async_fetch_worker PROPERTIES TIMEOUT 20)
if(AF_WITH_CURL)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(AF_CURL REQUIRED IMPORTED_TARGET "libcurl=${AF_CURL_VERSION}")
 add_library(async_fetch_curl STATIC curl_transport.c)
 target_link_libraries(async_fetch_curl PUBLIC async_fetch_worker PRIVATE PkgConfig::AF_CURL)
 af_warnings(async_fetch_curl)
 add_executable(test_async_fetch_curl test_curl.c)
 target_link_libraries(test_async_fetch_curl PRIVATE async_fetch_worker PkgConfig::AF_CURL)
 af_warnings(test_async_fetch_curl)
 add_test(NAME async_fetch_curl_offline COMMAND test_async_fetch_curl)
 set_tests_properties(async_fetch_curl_offline PROPERTIES TIMEOUT 10 SKIP_RETURN_CODE 77)
endif()
if(AF_BUILD_UI)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(AF_SDL REQUIRED IMPORTED_TARGET "sdl3=${AF_SDL_VERSION}")
 if(NOT TARGET cui_sdl3)
  set(CUI_BUILD_NATIVE ON CACHE BOOL "" FORCE)
  set(CUI_FETCH_SDL OFF CACHE BOOL "" FORCE)
  add_subdirectory(../.. cui-build EXCLUDE_FROM_ALL)
 endif()
 if(NOT TARGET cui_sdl3)
  message(FATAL_ERROR "SDL3, FreeType and HarfBuzz are required for AF_BUILD_UI")
 endif()
 add_executable(cui_async_fetch main.c)
 target_link_libraries(cui_async_fetch PRIVATE async_fetch_worker cui_sdl3 PkgConfig::AF_SDL)
 if(AF_WITH_CURL)
  target_link_libraries(cui_async_fetch PRIVATE async_fetch_curl)
  target_compile_definitions(cui_async_fetch PRIVATE AF_WITH_CURL=1)
 endif()
 af_warnings(cui_async_fetch)
 set(AF_TEST_FONT "${CUI_TEST_FONT}" CACHE FILEPATH "Font for private-display offline SDL smoke")
 if(AF_TEST_FONT AND EXISTS "${AF_TEST_FONT}")
  add_test(NAME async_fetch_sdl COMMAND cui_async_fetch --offline-smoke --font "${AF_TEST_FONT}"
    --capture "${CMAKE_CURRENT_BINARY_DIR}/async-fetch-smoke.bmp")
  set_tests_properties(async_fetch_sdl PROPERTIES TIMEOUT 25)
  add_test(NAME async_fetch_sdl_pending_shutdown COMMAND cui_async_fetch --offline-close-pending --font "${AF_TEST_FONT}")
  set_tests_properties(async_fetch_sdl_pending_shutdown PROPERTIES TIMEOUT 25)
 endif()
endif()
