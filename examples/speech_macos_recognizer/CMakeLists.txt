cmake_minimum_required(VERSION 3.20)
project(cui_speech_recognizer_example LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# No native window/device APIs are used by these independent fixtures.
set(CUI_BUILD_NATIVE OFF CACHE BOOL "Portable-only fixture test policy")
get_filename_component(CUI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT UNIX)
 message(FATAL_ERROR "This optional example currently supplies only a POSIX mutex (Linux/macOS)")
endif()
find_package(Threads REQUIRED)
add_library(speech_recognizer_host STATIC host.c host_lock.c)
target_include_directories(speech_recognizer_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CUI_ROOT}/include")
target_link_libraries(speech_recognizer_host PRIVATE Threads::Threads)
add_executable(test_host test_host.c "${CUI_ROOT}/src/base/cui_speech.c")
target_link_libraries(test_host PRIVATE speech_recognizer_host Threads::Threads)
if(NOT WIN32)
 target_link_libraries(test_host PRIVATE m)
endif()
foreach(target speech_recognizer_host test_host)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
foreach(case permissions transcript lifecycle bounds concurrent lifetime)
 add_test(NAME speech_recognizer_${case}_offline COMMAND test_host ${case})
 set_tests_properties(speech_recognizer_${case}_offline PROPERTIES TIMEOUT 15)
endforeach()

if(APPLE)
 enable_language(OBJC)
 find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
 find_library(SPEECH_FRAMEWORK Speech REQUIRED)
 find_library(AVFOUNDATION_FRAMEWORK AVFoundation REQUIRED)
 add_library(speech_recognizer_macos STATIC host_macos.m)
 target_link_libraries(speech_recognizer_macos PUBLIC speech_recognizer_host
   ${FOUNDATION_FRAMEWORK} ${SPEECH_FRAMEWORK} ${AVFOUNDATION_FRAMEWORK})
 add_executable(test_macos test_macos.m "${CUI_ROOT}/src/base/cui_speech.c")
 target_link_libraries(test_macos PRIVATE speech_recognizer_macos m)
 foreach(target speech_recognizer_macos test_macos)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror
    -mmacosx-version-min=13.0 "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc;-fblocks>")
  target_link_options(${target} PRIVATE -mmacosx-version-min=13.0)
 endforeach()
 add_test(NAME speech_recognizer_macos_injected COMMAND test_macos)
 set_tests_properties(speech_recognizer_macos_injected PROPERTIES TIMEOUT 15)
endif()
