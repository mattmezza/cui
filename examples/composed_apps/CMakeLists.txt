cmake_minimum_required(VERSION 3.20)
project(cui_composed_app LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_subdirectory(../.. cui-build)
add_library(composed_app STATIC app.c)
target_include_directories(composed_app PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(composed_app PUBLIC cui)
cui_warnings(composed_app)
add_executable(test_composed_app test_app.c)
target_link_libraries(test_composed_app PRIVATE composed_app)
cui_warnings(test_composed_app)
add_test(NAME composed_app_interactions COMMAND test_composed_app)
set_tests_properties(composed_app_interactions PROPERTIES TIMEOUT 30)
add_executable(northstar main.c)
target_link_libraries(northstar PRIVATE composed_app)
cui_warnings(northstar)
if(TARGET cui_sdl3)
 target_link_libraries(northstar PRIVATE cui_sdl3)
 target_compile_definitions(northstar PRIVATE APP_NATIVE=1)
endif()
