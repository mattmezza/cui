cmake_minimum_required(VERSION 3.20)
project(cui VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CUI_BUILD_NATIVE "Build optional SDL3 and shaped text adapters" ON)
option(CUI_SANITIZE "Enable ASan/UBSan" OFF)
option(CUI_FETCH_SDL "Fetch pinned SDL3 when system package absent" OFF)
if(CUI_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
function(cui_warnings target)
 if(target MATCHES "^(test_|dist_)")
  if(MSVC)
   target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
   target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
 endif()
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endfunction()
add_library(cui STATIC src/core/cui.c)
target_include_directories(cui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE third_party/clay)
cui_warnings(cui)
if(NOT MSVC)
 target_link_libraries(cui PUBLIC m)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/base/cui_base.c)

endif()
add_library(cui_software STATIC backends/software/cui_software.c)
target_link_libraries(cui_software PUBLIC cui)
cui_warnings(cui_software)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(PkgConfig QUIET)
if(CUI_BUILD_NATIVE AND PkgConfig_FOUND)
 pkg_check_modules(TEXT QUIET IMPORTED_TARGET freetype2 harfbuzz)
 pkg_check_modules(SDL3 QUIET IMPORTED_TARGET sdl3>=3.2)
 if(TEXT_FOUND)
  add_library(cui_text STATIC src/text/cui_text.c)
  target_link_libraries(cui_text PUBLIC cui PRIVATE PkgConfig::TEXT)
  cui_warnings(cui_text)
  target_compile_definitions(cui_software PUBLIC CUI_HAS_TEXT=1)
  target_link_libraries(cui_software PUBLIC cui_text)
 endif()
 if(NOT SDL3_FOUND AND CUI_FETCH_SDL)
  include(FetchContent)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 7f3ae3d57459e59943a4ecfefc8f6277ec6bf540 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(SDL)
  set(CUI_SDL_TARGET SDL3::SDL3)
 elseif(SDL3_FOUND)
  set(CUI_SDL_TARGET PkgConfig::SDL3)
 endif()
 if(CUI_SDL_TARGET AND TEXT_FOUND)
  add_library(cui_sdl3 STATIC backends/sdl3/cui_sdl3.c)
  target_link_libraries(cui_sdl3 PUBLIC cui cui_text PRIVATE ${CUI_SDL_TARGET})
  cui_warnings(cui_sdl3)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/gallery/main.c)
   add_executable(cui_gallery gallery/main.c)
   target_link_libraries(cui_gallery PRIVATE cui_sdl3 cui_software)
   cui_warnings(cui_gallery)
  endif()
 endif()
endif()

 message(STATUS "Shaped text and native gallery unavailable/disabled; portable core/software remain enabled")
endif()
