cmake_minimum_required(VERSION 3.20)
project(cui_bridge_private LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CUI_BUILD_BIDI OFF CACHE BOOL "" FORCE)
set(CUI_BUILD_CPP_EXAMPLE OFF CACHE BOOL "" FORCE)
set(CUI_GPU_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../.. ${CMAKE_CURRENT_BINARY_DIR}/cui_source EXCLUDE_FROM_ALL)
add_library(cui_bridge SHARED ../../src/bridge/cui_bridge.c)
target_compile_definitions(cui_bridge PRIVATE CUI_BRIDGE_BUILD=1)
target_include_directories(cui_bridge PUBLIC ../../include PRIVATE ../../src/bridge)
target_link_libraries(cui_bridge PRIVATE cui)
if(TARGET cui_sdl3)
 target_sources(cui_bridge PRIVATE ../../src/bridge/cui_bridge_sdl.c)
 target_compile_definitions(cui_bridge PRIVATE CUI_BRIDGE_WITH_SDL=1)
 target_link_libraries(cui_bridge PRIVATE cui_sdl3)
endif()
if(MSVC)
 target_compile_options(cui_bridge PRIVATE /W4)
else()
 target_compile_options(cui_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(test_bridge_core tests/test_bridge_core.c)
target_link_libraries(test_bridge_core PRIVATE cui_bridge)
if(MSVC)
 target_compile_options(test_bridge_core PRIVATE /W4)
else()
 target_compile_options(test_bridge_core PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
add_test(NAME bridge_core COMMAND test_bridge_core)
add_test(NAME bridge_python
 COMMAND ${CMAKE_COMMAND} -E env
  PYTHONPATH=${CMAKE_CURRENT_LIST_DIR}/../python
  CUI_BRIDGE_LIBRARY=$<TARGET_FILE:cui_bridge>
  ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../python/test_bridge.py)
