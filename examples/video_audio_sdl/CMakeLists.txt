cmake_minimum_required(VERSION 3.20)
project(video_audio_sdl LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIDEO_AUDIO_SANITIZE "ASan/UBSan" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl3=3.4.18)
find_package(Threads REQUIRED)
add_library(output STATIC output.c sdl.c ../video_audio_host/host.c)
target_link_libraries(output PUBLIC PkgConfig::SDL Threads::Threads m)
add_executable(test_output test_output.c)
target_link_libraries(test_output PRIVATE output)
foreach(t output test_output)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 if(VIDEO_AUDIO_SANITIZE)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
enable_testing()
foreach(case drain silence pause generations failures units teardown dummy unbound setup)
 add_test(NAME sdl_${case} COMMAND test_output ${case})
 set_tests_properties(sdl_${case} PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
endforeach()
