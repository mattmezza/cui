cmake_minimum_required(VERSION 3.20)
project(cui_control_events_example LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This optional example is currently validated on Linux only")
endif()
set(CE_SDL_VERSION "3.4.18" CACHE STRING "Exact validated installed SDL3 version; override requires revalidation")
find_package(PkgConfig REQUIRED)
pkg_check_modules(CE_SDL REQUIRED IMPORTED_TARGET "sdl3=${CE_SDL_VERSION}")
set(CUI_BUILD_NATIVE ON CACHE BOOL "" FORCE)
set(CUI_FETCH_SDL OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. cui-build EXCLUDE_FROM_ALL)
if(NOT TARGET cui_sdl3)
 message(FATAL_ERROR "Installed SDL3, FreeType and HarfBuzz are required")
endif()
add_library(control_events_app STATIC app.c)
target_include_directories(control_events_app PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(control_events_app PUBLIC cui)
add_executable(cui_control_events main.c)
target_link_libraries(cui_control_events PRIVATE control_events_app cui_sdl3 PkgConfig::CE_SDL)
add_executable(test_control_events_app test_app.c)
target_link_libraries(test_control_events_app PRIVATE control_events_app)
foreach(target control_events_app cui_control_events test_control_events_app)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endforeach()
enable_testing()
add_test(NAME control_events_example_model COMMAND test_control_events_app)
set(CE_TEST_FONT "${CUI_TEST_FONT}" CACHE FILEPATH "Font for private-display smoke")
if(CE_TEST_FONT AND EXISTS "${CE_TEST_FONT}")
 add_test(NAME control_events_example_sdl COMMAND cui_control_events --smoke --font "${CE_TEST_FONT}"
  --capture "${CMAKE_CURRENT_BINARY_DIR}/control-events.bmp")
 set_tests_properties(control_events_example_sdl PROPERTIES TIMEOUT 20)
endif()
