if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 include(root_bridge.cmake)
 return()
endif()
# Standalone validation harness; installation uses the shared root entry point.
cmake_minimum_required(VERSION 3.20)
project(cui_accessibility_harness LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CUI_SANITIZE "Enable C ASan/UBSan" OFF)
set(ACCESSKIT_CACHE "$ENV{HOME}/.cache/cui-accesskit" CACHE PATH "Pinned private dependency cache")
set(CUI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
file(READ "${CUI_ROOT}/third_party/accesskit/pin.json" PIN)
string(JSON ACCESSKIT_COMMIT GET "${PIN}" commit)
set(ACCESSKIT_SOURCE "${ACCESSKIT_CACHE}/accesskit-c-${ACCESSKIT_COMMIT}")
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3>=3.2)
pkg_check_modules(TEXT REQUIRED IMPORTED_TARGET freetype2 harfbuzz)
pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
add_library(cui STATIC "${CUI_ROOT}/src/core/cui.c" "${CUI_ROOT}/src/effects/cui_effects.c")
target_include_directories(cui PUBLIC "${CUI_ROOT}/include" PRIVATE "${CUI_ROOT}/third_party/clay")
target_link_libraries(cui PUBLIC m)
add_library(cui_accessibility STATIC cui_accessibility.c)
target_compile_definitions(cui_accessibility PRIVATE CUI_A11Y_TESTING=1)
target_include_directories(cui_accessibility PRIVATE "${ACCESSKIT_SOURCE}/include")
target_link_libraries(cui_accessibility PUBLIC cui Threads::Threads "${ACCESSKIT_CACHE}/target/release/libaccesskit.a" dl m)
add_executable(test_accessibility "${CUI_ROOT}/tests/unit/test_accessibility.c")
target_include_directories(test_accessibility PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(test_accessibility PRIVATE cui_accessibility)
add_library(cui_sdl3 STATIC "${CUI_ROOT}/backends/sdl3/cui_sdl3.c" "${CUI_ROOT}/src/text/cui_text.c")
target_link_libraries(cui_sdl3 PUBLIC cui PkgConfig::SDL3 PkgConfig::TEXT)
add_executable(test_accessibility_native "${CUI_ROOT}/tests/integration/test_accessibility_native.c")
target_include_directories(test_accessibility_native PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(test_accessibility_native PRIVATE cui_accessibility cui_sdl3 PkgConfig::SDL3 PkgConfig::ATSPI)
foreach(T cui cui_accessibility cui_sdl3 test_accessibility test_accessibility_native)
 target_compile_options(${T} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 if(CUI_SANITIZE)
  target_compile_options(${T} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${T} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
enable_testing()
add_test(NAME accessibility COMMAND test_accessibility)
# Runner provides a private DBus/Xvfb session; see NATIVE_ACCESSIBILITY.md.
add_test(NAME accessibility_native COMMAND "${CMAKE_CURRENT_LIST_DIR}/run_native.sh" $<TARGET_FILE:test_accessibility_native>)
set_tests_properties(accessibility_native PROPERTIES TIMEOUT 30 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
