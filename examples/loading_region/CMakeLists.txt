cmake_minimum_required(VERSION 3.20)
project(cui_loading_region LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(CUI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Reuse canonical source selection; build only explicitly requested targets.
set(CUI_BUILD_NATIVE OFF CACHE BOOL "" FORCE)
set(CUI_BUILD_BIDI OFF CACHE BOOL "" FORCE)
add_subdirectory("${CUI_ROOT}" cui EXCLUDE_FROM_ALL)
add_library(loading_scene STATIC scene.c)
target_link_libraries(loading_scene PUBLIC cui)
add_executable(loading_region main.c)
target_link_libraries(loading_region PRIVATE loading_scene)
add_executable(test_alert_loading_semantics "${CUI_ROOT}/tests/unit/test_alert_loading_semantics.c")
target_link_libraries(test_alert_loading_semantics PRIVATE loading_scene)
foreach(target loading_scene loading_region test_alert_loading_semantics)
 cui_warnings(${target})
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /UNDEBUG)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 endif()
endforeach()
enable_testing()
add_test(NAME alert_loading_semantics COMMAND test_alert_loading_semantics)
set_tests_properties(alert_loading_semantics PROPERTIES TIMEOUT 10)

option(LOADING_REGION_ACCESSKIT "Use existing pinned Linux AccessKit cache for real private-bus proof" OFF)
if(LOADING_REGION_ACCESSKIT)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Existing AccessKit package supports Linux only")
 endif()
 # Reuse canonical checksum verification and optional dependency setup.
 add_subdirectory("${CUI_ROOT}/backends/accesskit" accesskit EXCLUDE_FROM_ALL)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
 add_executable(test_loading_native test_native.c)
 target_link_libraries(test_loading_native PRIVATE loading_scene cui_accessibility PkgConfig::ATSPI)
 cui_warnings(test_loading_native)
 target_compile_options(test_loading_native PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 add_test(NAME alert_loading_native COMMAND "${CUI_ROOT}/backends/accesskit/run_native.sh" $<TARGET_FILE:test_loading_native>)
 set_tests_properties(alert_loading_native PROPERTIES TIMEOUT 30 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endif()
