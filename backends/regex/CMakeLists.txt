# Optional prebuilt archive consumer harness. Root CMake registration is separate.
cmake_minimum_required(VERSION 3.20)
project(cui_regex_consumer LANGUAGES C)
include(CTest)
get_filename_component(CUI_REGEX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CUI_REGEX_LIBRARY "" CACHE FILEPATH "Pinned Rust1.85.1 native libCUI regex archive")
set(CUI_REGEX_CORE_LIBRARY "" CACHE FILEPATH "Matching portable CUI archive")
if(NOT EXISTS "${CUI_REGEX_LIBRARY}" OR NOT EXISTS "${CUI_REGEX_CORE_LIBRARY}")
  message(FATAL_ERROR "Provide both prebuilt archives; this optional harness never invokes Rust")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This native link receipt is Linux-specific; other targets need their own rustc native-static-libs receipt")
endif()
find_package(Threads REQUIRED)
add_library(cui_regex_core STATIC IMPORTED)
set_target_properties(cui_regex_core PROPERTIES IMPORTED_LOCATION "${CUI_REGEX_CORE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CUI_REGEX_ROOT}/include")
add_library(cui_regex STATIC IMPORTED)
add_library(cui::regex ALIAS cui_regex)
set_target_properties(cui_regex PROPERTIES IMPORTED_LOCATION "${CUI_REGEX_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CUI_REGEX_ROOT}/include" INTERFACE_LINK_LIBRARIES "cui_regex_core;gcc_s;util;rt;Threads::Threads;m;${CMAKE_DL_LIBS}")
foreach(name IN ITEMS test_input_matcher test_regex_provider)
  add_executable(${name} "${CUI_REGEX_ROOT}/tests/unit/${name}.c")
  target_compile_features(${name} PRIVATE c_std_11)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(${name} PRIVATE cui::regex)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
add_executable(test_input_regex_control "${CUI_REGEX_ROOT}/tests/integration/test_input_regex_control.c")
target_compile_features(test_input_regex_control PRIVATE c_std_11)
target_compile_options(test_input_regex_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(test_input_regex_control PRIVATE cui::regex)
add_test(NAME test_input_regex_control COMMAND test_input_regex_control)
