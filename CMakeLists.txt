cmake_minimum_required(VERSION 3.20)
project(cui VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(cui STATIC src/core/cui.c)
target_include_directories(cui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE third_party/clay)
if(NOT MSVC)
 target_compile_options(cui PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(cui PUBLIC m)
endif()
add_executable(hello_world examples/hello_world/main.c)
target_link_libraries(hello_world PRIVATE cui)
enable_testing()
add_executable(test_bootstrap tests/unit/test_bootstrap.c)
target_link_libraries(test_bootstrap PRIVATE cui)
add_test(NAME bootstrap COMMAND test_bootstrap)
install(TARGETS cui ARCHIVE DESTINATION lib)
install(DIRECTORY include/cui DESTINATION include)
